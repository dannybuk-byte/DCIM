### To Add "Document Evidence" Buttons:
You can manually add this to facility detail modals:

```tsx
import { useEvidence } from '../hooks/useEvidence';

function FacilityModal({ facility }) {
  const { createPackage } = useEvidence();
  
  const handleDocumentEvidence = async () => {
    await createPackage(
      facility,
      [window.location.href, 'https://sec.gov', 'https://epa.gov'],
      'manual'
    );
  };
  
  return (
    <div>
      {/* ... facility details ... */}
      <button onClick={handleDocumentEvidence}>
        📋 Document Evidence
      </button>
    </div>
  );
}
```
